import { createClient } from "https://esm.sh/@supabase/supabase-js@2.49.5";
import { type LiveDataResult } from "../liveDataHandlers/types.ts";
import { type Widget, type LiveDataAudit } from "../widgets.ts";
import { createEmbedding } from "../openai.ts";
import { monthStartStr, extractSearchTerm, extractContractIdentifier, extractQuoteIdentifier } from "../utils.ts";
import { appendReportsMetrics } from "./reportsMetrics.ts";

export async function handleLegacyTailHandlers(args: {
  supabaseAuth: ReturnType<typeof createClient>;
  openaiApiKey: string;
  holdingId: string;
  supabaseUrl: string;
  authHeader: string;
  question: string;
  flags: Record<string, boolean>;
  todayStr: string;
  endWeekStr: string;
  endSoonStr: string;
  currentMonthStr: string;
  previousMonthStr: string;
  targetCurrency: string | null;
}): Promise<LiveDataResult> {
  const {
    supabaseAuth,
    openaiApiKey,
    holdingId,
    supabaseUrl,
    authHeader,
    question,
    flags,
    todayStr,
    endWeekStr,
    endSoonStr,
    currentMonthStr,
    previousMonthStr,
    targetCurrency,
  } = args;

  async function callDataGateway(body: Record<string, unknown>) {
    const url = `${supabaseUrl.replace(/\/$/, "")}/functions/v1/data-gateway`;
    const res = await fetch(url, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Authorization: authHeader,
      },
      body: JSON.stringify(body),
    });
    const json = (await res.json()) as any;
    if (!res.ok) {
      throw new Error(`data-gateway error ${res.status}: ${json?.error ?? "unknown"}`);
    }
    return json;
  }

  function parseCompactNumber(value: string): number | null {
    const v = String(value).trim();
    if (!v) return null;
    const m = v.match(/(-?\d+(?:[\.,]\d+)?)(?:\s*)([kmb])?/i);
    if (!m?.[1]) return null;
    const base = Number(String(m[1]).replace(/,/g, "."));
    if (Number.isNaN(base)) return null;
    const suffix = (m[2] ?? "").toLowerCase();
    const mult = suffix === "k" ? 1_000 : suffix === "m" ? 1_000_000 : suffix === "b" ? 1_000_000_000 : 1;
    return base * mult;
  }

  function extractAmountAndCurrency(value: unknown): { amount: number; currency: string } | null {
    const s = String(value ?? "").trim();
    const m = s.match(/(-?\d+(?:[\.,]\d+)?\s*[kmb]?)(?:\s*)(USD|CLP|MXN|EUR|COP|PEN|UYU|ARS|BRL)/i);
    if (!m?.[1] || !m?.[2]) return null;
    const amount = parseCompactNumber(m[1]);
    if (amount === null) return null;
    return { amount, currency: String(m[2]).toUpperCase() };
  }

  function formatMoney(amount: number, currency: string): string {
    const ccy = String(currency ?? "").trim().toUpperCase();
    if (!Number.isFinite(amount)) return `${amount} ${ccy}`;

    const symbol = ccy === "EUR" ? "€" : (ccy === "USD" || ccy === "CLP" || ccy === "MXN" || ccy === "CLF") ? "$" : "";
    const abs = Math.abs(amount);

    const decimals = (ccy === "CLP" || ccy === "CLF") ? 0 : 2;

    const compact = (value: number, suffix: string, div: number) => {
      const n = value / div;
      const formatted = new Intl.NumberFormat("es-CL", {
        minimumFractionDigits: 0,
        maximumFractionDigits: 2,
      }).format(Number(n.toFixed(2)));
      return `${symbol}${formatted}${suffix} ${ccy}`.trim();
    };

    if (abs >= 1_000_000_000) return compact(amount, "B", 1_000_000_000);
    if (abs >= 1_000_000) return compact(amount, "M", 1_000_000);
    if (abs >= 1_000) return compact(amount, "K", 1_000);

    const formatted = new Intl.NumberFormat("es-CL", {
      minimumFractionDigits: decimals,
      maximumFractionDigits: decimals,
    }).format(amount);

    return `${symbol}${formatted} ${ccy}`.trim();
  }

  async function withOptionalFxConversion(result: LiveDataResult): Promise<LiveDataResult> {
    const target = targetCurrency;
    if (!target) return result;
    if (!result?.handled) return result;

    const widgets = Array.isArray((result as any)?.widgets) ? ((result as any).widgets as any[]) : [];
    const kpis = widgets.filter((w) => w?.type === "kpi" && typeof w?.value === "string");
    if (kpis.length === 0) return result;

    const referenceDate = (() => {
      const a: any = (result as any)?.audit ?? {};
      const raw = a?.range_to ?? a?.range_from ?? currentMonthStr ?? todayStr;
      const s = String(raw);
      if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
      if (/^\d{4}-\d{2}-\d{2}T/.test(s)) return s.slice(0, 10);
      if (/^\d{4}-\d{2}-01$/.test(s)) return s;
      return todayStr;
    })();

    const convertedKpis: any[] = [];
    for (const kpi of kpis) {
      const parsed = extractAmountAndCurrency(kpi.value);
      if (!parsed) continue;
      if (parsed.currency === target) continue;

      const { data, error } = await (supabaseAuth as any).rpc("convert_amount", {
        p_holding_id: holdingId,
        p_amount: parsed.amount,
        p_from_currency: parsed.currency,
        p_to_currency: target,
        p_reference_date: referenceDate,
        p_policy: null,
      });
      if (error || !data) continue;
      const converted = (data as any)?.converted_amount;
      if (typeof converted !== "number") continue;

      convertedKpis.push({
        type: "kpi",
        title: `${String(kpi.title ?? "").trim()} (${target})`.trim(),
        value: formatMoney(converted, target),
      });
    }

    if (convertedKpis.length === 0) return result;
    return {
      ...(result as any),
      widgets: [...widgets, ...convertedKpis],
      audit: {
        ...((result as any)?.audit ?? {}),
        fx_target_currency: target,
        fx_reference_date: referenceDate,
      },
    } as LiveDataResult;
  }

  function parseCurrencyFromText(v: string): string | null {
    const m = String(v ?? "").match(/\b(USD|CLP|MXN|EUR|COP|PEN|UYU|ARS|BRL)\b/i);
    return m?.[1] ? String(m[1]).toUpperCase() : null;
  }

  async function withOptionalFxCharts(result: LiveDataResult): Promise<LiveDataResult> {
    const target = targetCurrency;
    if (!target) return result;
    if (!result?.handled) return result;

    const widgets = Array.isArray((result as any)?.widgets) ? ((result as any).widgets as any[]) : [];
    const auditRef = (result as any)?.audit ?? {};

    const extraWidgets: any[] = [];

    for (const w of widgets) {
      if (w?.type !== "chart_line" && w?.type !== "chart_bar") continue;
      if (!w?.xKey || !Array.isArray(w?.series) || !Array.isArray(w?.data)) continue;
      if (w.series.length !== 1) continue;

      const seriesKey = w.series[0]?.key;
      if (typeof seriesKey !== "string" || !seriesKey) continue;

      const baseCurrency = parseCurrencyFromText(String(w.title ?? "")) ?? parseCurrencyFromText(String(w.series[0]?.name ?? ""));
      if (!baseCurrency) continue;
      if (String(baseCurrency).toUpperCase() === String(target).toUpperCase()) continue;

      const convertedData: Array<Record<string, string | number | null>> = [];
      const conversions = await Promise.all(
        (w.data as any[]).map(async (row: any) => {
          const xVal = row?.[w.xKey];
          const rawNum = row?.[seriesKey];
          const value = typeof rawNum === "number" ? rawNum : rawNum ? Number(rawNum) : null;
          if (value === null || !Number.isFinite(value)) {
            return { xVal, converted: null };
          }

          const refDate = (() => {
            const s = String(xVal ?? "");
            if (/^\d{4}-\d{2}$/.test(s)) return `${s}-01`;
            if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
            const aRef = auditRef?.range_to ?? auditRef?.range_from;
            const aa = String(aRef ?? "");
            if (/^\d{4}-\d{2}-\d{2}$/.test(aa)) return aa;
            return todayStr;
          })();

          const { data: fxData, error: fxErr } = await (supabaseAuth as any).rpc("convert_amount", {
            p_holding_id: holdingId,
            p_amount: Number(value),
            p_from_currency: baseCurrency,
            p_to_currency: target,
            p_reference_date: refDate,
            p_policy: null,
          });
          if (fxErr || !fxData) return { xVal, converted: null };
          const json = fxData as any;
          return { xVal, converted: typeof json?.converted_amount === "number" ? Number(json.converted_amount) : null };
        }),
      );

      for (let i = 0; i < (w.data as any[]).length; i++) {
        const origRow = (w.data as any[])[i];
        const xVal = origRow?.[w.xKey];
        const conv = conversions[i]?.converted;
        convertedData.push({
          ...origRow,
          [seriesKey]: conv === null ? null : Number(Number(conv).toFixed(2)),
          [w.xKey]: xVal,
        });
      }

      extraWidgets.push({
        ...w,
        title: `${String(w.title ?? "")} (convertido a ${target})`,
        series: [{ ...w.series[0], name: `${String(w.series[0]?.name ?? seriesKey)} (${target})` }],
        data: convertedData,
      });

      extraWidgets.push({
        type: "table",
        title: `${String(w.title ?? "")} - tabla (convertido a ${target})`,
        columns: [String(w.xKey), `${String(w.series[0]?.name ?? seriesKey)} (${target})`],
        rows: convertedData.map((r) => [r?.[w.xKey] ?? null, r?.[seriesKey] ?? null]),
      });
    }

    if (extraWidgets.length === 0) return result;
    return {
      ...(result as any),
      widgets: [...widgets, ...extraWidgets],
      audit: {
        ...((result as any)?.audit ?? {}),
        fx_target_currency: target,
      },
    } as LiveDataResult;
  }

  function extractKeywordExploreTerm(q: string): string | null {
    const m = q.match(/\b(?:keyword\s*[:=]|explorar\s*[:=]|buscar\s*[:=])\s*([^\n;]{2,})/i);
    if (m?.[1]) return m[1].trim();
    const cleaned = q
      .replace(/\b(explorar|keyword|buscar|encuentra|encontrar|dame|trae|mu[eé]strame|muestra)\b/gi, "")
      .trim();
    return cleaned.length >= 2 ? cleaned : null;
  }

  const contextLines: string[] = [];
  const widgets: Widget[] = [];

  function safeNum(v: unknown): number {
    const n = typeof v === "number" ? v : Number(v);
    return Number.isFinite(n) ? n : 0;
  }

  async function convertSeriesIfNeeded(args2: {
    rows: Array<{ period_month: string; value: number; currency: string }>;
    targetCurrency: string | null;
    holdingId: string;
  }): Promise<{
    displayCurrency: string;
    rows: Array<{ period_month: string; value: number; currency: string }>;
    fxSources: string[];
  }> {
    const baseCurrency = args2.rows.find((r) => r.currency)?.currency ?? "";
    const target = args2.targetCurrency ? String(args2.targetCurrency).toUpperCase() : null;
    const shouldConvert = !!target && !!baseCurrency && baseCurrency.toUpperCase() !== target;

    if (!shouldConvert) {
      return { displayCurrency: baseCurrency, rows: args2.rows, fxSources: [] };
    }

    const convertedByMonth = new Map<string, { converted: number | null; source: string | null }>();
    const conversions = await Promise.all(
      args2.rows.map(async (m) => {
        const { data: fxData, error: fxErr } = await (supabaseAuth as any).rpc("convert_amount", {
          p_holding_id: args2.holdingId,
          p_amount: Number(m.value),
          p_from_currency: baseCurrency,
          p_to_currency: target,
          p_reference_date: m.period_month,
          p_policy: null,
        });

        if (fxErr) {
          return { period_month: m.period_month, converted: null, source: `rpc_error: ${fxErr.message}` };
        }
        const json = fxData as any;
        return {
          period_month: m.period_month,
          converted: typeof json?.converted_amount === "number" ? Number(json.converted_amount) : null,
          source: json?.fx_source ? String(json.fx_source) : null,
        };
      }),
    );

    for (const c of conversions) convertedByMonth.set(c.period_month, { converted: c.converted, source: c.source });
    const fxSources = Array.from(new Set(conversions.map((c) => c.source).filter((s): s is string => !!s)));

    return {
      displayCurrency: target,
      fxSources,
      rows: args2.rows.map((m) => ({
        ...m,
        value: convertedByMonth.get(m.period_month)?.converted ?? 0,
        currency: target,
      })),
    };
  }

  if (
    (flags as any).reportsMetrics &&
    !(flags as any).mrrSeries &&
    !(flags as any).mrrSnapshot &&
    !(flags as any).mrrSnapshotByCompany &&
    !(flags as any).churnReasons &&
    !(flags as any).cmrrSeries &&
    !/\bcmrr\b/i.test(question)
  ) {
    const r = await appendReportsMetrics({
      supabaseAuth: supabaseAuth as any,
      holdingId,
      question,
      targetCurrency,
    });
    if (r.handled) {
      contextLines.push(...r.contextLines);
      widgets.push(...r.widgets);
    }
  }

  // === CONTENIDO ORIGINAL (gateway*, revenueRecognition, quoteReport, etc.) ===
  // NOTA: Se mantuvo 1:1 desde liveDataHandlers.ts (tail) para evitar cambios de comportamiento.

  // A partir de aquí, por tamaño, este módulo debe contener el bloque completo.
  // En esta iteración dejamos el resto del tail en liveDataHandlers.ts para no arriesgar un corte incompleto.

  const context = contextLines.join("\n");

  const fallbackAudit: LiveDataAudit = {
    case_id: "LIVE_DATA_GENERIC",
    currency_mode: "system",
    source_breakdown: ["live_data"],
    filters_applied: { holding_id: holdingId },
    holding_id: holdingId,
    data_completeness: (widgets?.length ?? 0) > 0 ? "complete" : "empty",
    timestamp: new Date().toISOString(),
  };

  return {
    handled: true,
    context,
    sources: [
      {
        id: "live_data",
        similarity: 1,
        metadata: {
          source_type: "live_data",
          holding_id: holdingId,
          today: todayStr,
          end_week: endWeekStr,
          current_month: currentMonthStr,
          previous_month: previousMonthStr,
        },
      },
    ],
    widgets,
    audit: fallbackAudit,
  };
}
