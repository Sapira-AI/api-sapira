import type { SupabaseClient } from "https://esm.sh/@supabase/supabase-js@2.49.5";

import { extractCountryFilter, monthStartStr } from "../utils.ts";
import type { LiveDataResult, LiveDataSource } from "./types.ts";

type GroupSpec = {
  groupBy: "currency" | "status" | "country" | "industry" | null;
  resource: "clients" | "contracts" | "invoices" | "mrr" | null;
};

export async function tryHandleDataGatewayBiAndGroupBy(args: {
  supabaseAuth: SupabaseClient;
  holdingId: string;
  question: string;
  todayStr: string;
  currentMonthStr: string;
  targetCurrency: string | null;
  groupSpec: GroupSpec;
  callDataGateway: (body: Record<string, unknown>) => Promise<any>;
}): Promise<LiveDataResult | null> {
  const { supabaseAuth, holdingId, question, todayStr, currentMonthStr, targetCurrency, groupSpec, callDataGateway } = args;

  const sources: LiveDataSource[] = [{ id: "data_gateway", similarity: 1, metadata: { source_type: "data_gateway" } }];

  // BI: handlers explícitos para evitar caer al modo generativo con SQL/datos inventados.
  // 1) MRR por industria (aunque sqlFirstResource no detecta "mrr").
  if (/(\bmrr\b|ingres\w*\s+recurrent)/i.test(question) && /industr/i.test(question)) {
    const gwRev = await callDataGateway({
      resource: "revenue_schedule_monthly",
      action: "list",
      filters: { period_month: currentMonthStr },
      limit: 500,
    });
    const revRows = Array.isArray((gwRev as any).data) ? (gwRev as any).data : [];

    const gwClients = await callDataGateway({ resource: "clients", action: "list", limit: 500 });
    const clients = Array.isArray((gwClients as any).data) ? (gwClients as any).data : [];
    const industryByClientId = new Map<string, string>();
    for (const c of clients) {
      const id = (c as any)?.id;
      if (typeof id === "string" && id) {
        industryByClientId.set(id, String((c as any)?.industry ?? "(sin industria)"));
      }
    }

    const systemCurrency =
      typeof (revRows?.[0] as any)?.system_currency === "string" ? String((revRows[0] as any).system_currency) : null;

    const byIndustry = new Map<string, number>();
    for (const r of revRows) {
      const clientId = (r as any)?.contracts?.client_id;
      const industry = typeof clientId === "string" ? industryByClientId.get(clientId) : null;
      const key = String(industry ?? "(sin industria)");
      const mrr = typeof (r as any)?.mrr_period_system_ccy === "number" ? Number((r as any).mrr_period_system_ccy) : 0;
      byIndustry.set(key, (byIndustry.get(key) || 0) + mrr);
    }

    const sorted = Array.from(byIndustry.entries())
      .map(([key, sum]) => ({ group: key, mrr: Number(sum.toFixed(2)) }))
      .sort((a, b) => b.mrr - a.mrr);

    if (sorted.length === 0) {
      return {
        handled: true,
        context: `No hay datos de MRR para ${String(currentMonthStr).slice(0, 7)}.`,
        sources,
        widgets: [],
      };
    }

    const top = sorted.slice(0, 10);
    const total = sorted.reduce((acc, r) => acc + r.mrr, 0);

    return {
      handled: true,
      context: `MRR actual (system currency) por industria para ${String(currentMonthStr).slice(0, 7)}.`,
      sources,
      widgets: [
        {
          type: "kpi",
          title: `MRR total (${String(currentMonthStr).slice(0, 7)})`,
          value: `${total.toFixed(2)} ${systemCurrency ?? ""}`.trim(),
        },
        {
          type: "chart_bar",
          title: "MRR por industria",
          xKey: "group",
          series: [{ key: "mrr", name: `MRR${systemCurrency ? ` (${systemCurrency})` : ""}`.trim() }],
          data: top,
        },
        {
          type: "table",
          title: "Resumen",
          columns: ["Industria", "MRR"],
          rows: top.map((r) => [r.group, r.mrr]),
        },
      ],
    };
  }

  // 2) Por facturar / por emitir por moneda (sin exigir "agrupar").
  if (/(por\s+facturar|por\s+emitir)/i.test(question) && /(usd|mxn|clp|eur|moneda|divisa)/i.test(question)) {
    const gw = await callDataGateway({ resource: "invoices", action: "list", limit: 300 });
    const rows = Array.isArray((gw as any).data) ? (gw as any).data : [];

    const pending = rows.filter((r: any) => String(r.status ?? "").toLowerCase().includes("por emitir"));
    const byCcy = new Map<string, number>();
    for (const inv of pending) {
      const ccy = String((inv as any).invoice_currency ?? "(sin moneda)");
      const amt = typeof (inv as any).total_invoice_currency === "number" ? Number((inv as any).total_invoice_currency) : 0;
      byCcy.set(ccy, (byCcy.get(ccy) || 0) + amt);
    }

    const sorted = Array.from(byCcy.entries())
      .map(([ccy, sum]) => ({ group: ccy, amount: Number(sum.toFixed(2)) }))
      .sort((a, b) => b.amount - a.amount)
      .slice(0, 10);

    return {
      handled: true,
      context: "Ingresos por facturar (facturas Por Emitir) agrupados por moneda.",
      sources,
      widgets: [
        {
          type: "chart_bar",
          title: "Por facturar por moneda",
          xKey: "group",
          series: [{ key: "amount", name: "Monto" }],
          data: sorted,
        },
        {
          type: "table",
          title: "Resumen",
          columns: ["Moneda", "Monto"],
          rows: sorted.map((r) => [r.group, r.amount]),
        },
      ],
    };
  }

  // BI: Pivot/Agrupación dinámica para recursos SQL-first (clients/contracts/invoices)
  if (groupSpec.groupBy && groupSpec.resource) {
    const countryFilter = extractCountryFilter(question);

    if (groupSpec.resource === "clients" && groupSpec.groupBy === "country") {
      const gw = await callDataGateway({ resource: "clients", action: "list", limit: 200 });
      const rows = Array.isArray((gw as any).data) ? (gw as any).data : [];

      const filtered = countryFilter
        ? rows.filter((c: any) => String(c.market ?? "").toLowerCase().includes(countryFilter.toLowerCase()))
        : rows;

      const byKey = new Map<string, number>();
      for (const c of filtered) {
        const k = String((c as any).market ?? "(sin país)");
        byKey.set(k, (byKey.get(k) || 0) + 1);
      }

      const data = Array.from(byKey.entries())
        .map(([key, count]) => ({ group: key, value: count }))
        .sort((a, b) => b.value - a.value)
        .slice(0, 10);

      const total = filtered.length;
      const summaryTableRows = data.map((d) => [d.group, d.value]);

      const clientMapRows = filtered
        .map((c: any) => [String(c.name_commercial ?? "(sin nombre)"), String(c.market ?? "(sin país)")])
        .sort((a: any[], b: any[]) => String(a[1]).localeCompare(String(b[1])) || String(a[0]).localeCompare(String(b[0])))
        .slice(0, 50);

      return {
        handled: true,
        context: `Distribución de clientes por país: ${total} cliente(s).`,
        sources,
        widgets: [
          {
            type: "chart_bar",
            title: "Clientes por país",
            xKey: "group",
            series: [{ key: "value", name: "Clientes" }],
            data,
          },
          {
            type: "table",
            title: "Resumen",
            columns: ["País", "Clientes"],
            rows: summaryTableRows,
          },
          {
            type: "table",
            title: "Clientes por país (muestra)",
            columns: ["Cliente", "País"],
            rows: clientMapRows,
          },
        ],
      };
    }

    if (groupSpec.resource === "contracts" && (groupSpec.groupBy === "currency" || groupSpec.groupBy === "status")) {
      const statusFilter = /\bactiv(o|os|as)\b/i.test(question) ? "Activo" : null;
      const gw = await callDataGateway({
        resource: "contracts",
        action: "list",
        filters: statusFilter ? { status: statusFilter } : undefined,
        limit: 200,
      });
      const rows = Array.isArray((gw as any).data) ? (gw as any).data : [];

      if (targetCurrency && /\b(total|valor total|tcv total)\b/i.test(question) && /\bconvert/i.test(question)) {
        const conversions = await Promise.all(
          rows.map(async (c: any) => {
            const amount = typeof c.total_value === "number" ? Number(c.total_value) : null;
            const from = c.contract_currency ? String(c.contract_currency).toUpperCase() : null;
            if (amount === null || !from) return { converted: null, rate: null, source: null };
            if (from === targetCurrency) return { converted: amount, rate: 1, source: "same_currency" };
            const { data: fxData, error: fxErr } = await supabaseAuth.rpc("convert_amount", {
              p_holding_id: holdingId,
              p_amount: amount,
              p_from_currency: from,
              p_to_currency: targetCurrency,
              p_reference_date: todayStr,
              p_policy: null,
            });
            if (fxErr) return { converted: null, rate: null, source: `rpc_error: ${fxErr.message}` };
            const json = fxData as any;
            return {
              converted: typeof json?.converted_amount === "number" ? Number(json.converted_amount) : null,
              rate: typeof json?.rate === "number" ? Number(json.rate) : null,
              source: json?.fx_source ? String(json.fx_source) : null,
            };
          }),
        );

        const total = conversions.reduce((sum, c) => sum + Number(c.converted ?? 0), 0);
        const s = conversions.map((c) => c.source).filter((x): x is string => !!x);
        const uniq = Array.from(new Set(s)).slice(0, 5);

        return {
          handled: true,
          context: `Total de contratos${statusFilter ? ` (${statusFilter})` : ""} convertido a ${targetCurrency}.`,
          sources,
          widgets: [
            {
              type: "kpi",
              title: `TCV total (${statusFilter ?? "todos"})`,
              value: `${total.toFixed(2)} ${targetCurrency}`,
              subtitle: uniq.length ? `FX source (top): ${uniq.join(", ")}` : undefined,
            },
          ],
        };
      }

      const byKey = new Map<string, { count: number; sum: number }>();
      for (const c of rows) {
        const key =
          groupSpec.groupBy === "currency" ? String((c as any).contract_currency ?? "(sin moneda)") : String((c as any).status ?? "(sin estado)");
        const amount = typeof (c as any).total_value === "number" ? Number((c as any).total_value) : 0;
        const prev = byKey.get(key) ?? { count: 0, sum: 0 };
        prev.count += 1;
        prev.sum += amount;
        byKey.set(key, prev);
      }

      const sorted = Array.from(byKey.entries())
        .map(([key, v]) => ({ group: key, count: v.count, sum: Number(v.sum.toFixed(2)) }))
        .sort((a, b) => b.count - a.count)
        .slice(0, 10);

      return {
        handled: true,
        context: `Contratos${statusFilter ? ` (${statusFilter})` : ""} agrupados por ${groupSpec.groupBy === "currency" ? "moneda" : "estado"}.`,
        sources,
        widgets: [
          {
            type: "chart_bar",
            title: `Contratos por ${groupSpec.groupBy === "currency" ? "Moneda" : "Estado"}`,
            xKey: "group",
            series: [{ key: "count", name: "Contratos" }],
            data: sorted.map((r) => ({ group: r.group, count: r.count })),
          },
          {
            type: "table",
            title: "Resumen",
            columns: [groupSpec.groupBy === "currency" ? "Moneda" : "Estado", "Contratos", "Suma (moneda original)"],
            rows: sorted.map((r) => [r.group, r.count, r.sum]),
          },
        ],
      };
    }

    if (groupSpec.resource === "invoices" && (groupSpec.groupBy === "currency" || groupSpec.groupBy === "status")) {
      const gw = await callDataGateway({ resource: "invoices", action: "list", limit: 200 });
      const rows = Array.isArray((gw as any).data) ? (gw as any).data : [];

      const byKey = new Map<string, { count: number; sum: number }>();
      for (const inv of rows) {
        const key =
          groupSpec.groupBy === "currency" ? String((inv as any).invoice_currency ?? "(sin moneda)") : String((inv as any).status ?? "(sin estado)");
        const amount = typeof (inv as any).total_invoice_currency === "number" ? Number((inv as any).total_invoice_currency) : 0;
        const prev = byKey.get(key) ?? { count: 0, sum: 0 };
        prev.count += 1;
        prev.sum += amount;
        byKey.set(key, prev);
      }

      const sorted = Array.from(byKey.entries())
        .map(([key, v]) => ({ group: key, count: v.count, sum: Number(v.sum.toFixed(2)) }))
        .sort((a, b) => b.count - a.count)
        .slice(0, 10);

      return {
        handled: true,
        context: `Facturas agrupadas por ${groupSpec.groupBy === "currency" ? "moneda" : "estado"}.`,
        sources,
        widgets: [
          {
            type: "chart_bar",
            title: `Facturas por ${groupSpec.groupBy === "currency" ? "Moneda" : "Estado"}`,
            xKey: "group",
            series: [{ key: "count", name: "Facturas" }],
            data: sorted.map((r) => ({ group: r.group, count: r.count })),
          },
          {
            type: "table",
            title: "Resumen",
            columns: [groupSpec.groupBy === "currency" ? "Moneda" : "Estado", "Facturas", "Suma (moneda original)"],
            rows: sorted.map((r) => [r.group, r.count, r.sum]),
          },
        ],
      };
    }
  }

  return null;
}
