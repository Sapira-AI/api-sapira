export interface HandlerContext {
  supabase: SupabaseClient;
  question: string;
  holdingId: string;
  openaiApiKey: string;
  supabaseUrl: string;
  authHeader: string;
}

export interface HandlerRegistration {
  flag: keyof LiveDataFlags;
  handler: (context: HandlerContext) => Promise<LiveDataResult>;
  priority: number; // Para manejar orden de ejecución
}

/**
 * Registro de todos los handlers disponibles
 * Organizados por prioridad (menor número = mayor prioridad)
 */
const HANDLER_REGISTRY: HandlerRegistration[] = [
  // Handlers QA originales (alta prioridad)

/**
 * Ejecuta el handler apropiado basado en los flags activos
 */
export async function routeToHandler(
  flags: LiveDataFlags,
  context: HandlerContext
): Promise<LiveDataResult | null> {
  
  // Ordenar handlers por prioridad
  const sortedHandlers = HANDLER_REGISTRY.sort((a, b) => a.priority - b.priority);
  
  // Buscar el primer handler que tenga su flag activo
  for (const registration of sortedHandlers) {
    if (flags[registration.flag]) {
      try {
        console.log(`Ejecutando handler para flag: ${registration.flag}`);
        return await registration.handler(context);
      } catch (error) {
        console.error(`Error en handler ${registration.flag}:`, error);
        // Continuar con el siguiente handler si hay error
        continue;
      }
    }
  }

  // Handlers especiales que requieren lógica adicional
  if (flags.bookingToFirstInvoiceLeadTime) {
    if (isQa10BookingFacturaQuestion(context.question)) {
      return await handleQa10BookingToFirstInvoiceLeadTimeFlag(
        context.supabase,
        context.question,
        context.holdingId
      );
    } else {
      return await handleQa10BookingToFirstInvoiceLeadTimeText(
        context.supabase,
        context.question,
        context.holdingId
      );
    }
  }

  return null;
}

/**
 * Verifica si algún flag está activo
 */
export function hasActiveFlags(flags: LiveDataFlags): boolean {
  return HANDLER_REGISTRY.some(registration => flags[registration.flag]) ||
         flags.bookingToFirstInvoiceLeadTime;
}

/**
 * Obtiene la lista de flags activos para debugging
 */
export function getActiveFlags(flags: LiveDataFlags): string[] {
  const active: string[] = [];
  
  for (const registration of HANDLER_REGISTRY) {
    if (flags[registration.flag]) {
      active.push(registration.flag);
    }
  }
  
  if (flags.bookingToFirstInvoiceLeadTime) {
    active.push('bookingToFirstInvoiceLeadTime');
  }
  
  return active;
}
