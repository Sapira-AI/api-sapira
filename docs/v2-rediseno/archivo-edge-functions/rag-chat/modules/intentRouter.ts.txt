/**
 * Detecta si una consulta es conceptual (definiciones/reglas)
 * Implementa routing según especificación técnica
 */
export function isConceptualQuery(question: string): boolean {
  const q = question.toLowerCase();
  
  const conceptualPatterns = [
    "qué es", "que es", "qué significa", "que significa",
    "cómo se calcula", "como se calcula", "por qué", "por que",
    "cuál es la diferencia", "cual es la diferencia",
    "definición", "definicion", "concepto", "regla", "reglas",
    "política", "politica", "criterio", "criterios",
    "explica", "explicar", "diferencia entre"
  ];
  
  return conceptualPatterns.some(pattern => q.includes(pattern));
}

  const timePatterns = [
    "últimos", "ultimo", "última", "ultima", "pasados", "anteriores",
    "histórico", "historico", "tendencia", "evolución", "evolucion",

    "12 meses", "6 meses", "último año", "ultimo año",
    "serie", "series", "waterfall", "momentum", "movimiento", "movimientos"
  ];
  
  // Métricas expandidas que siempre requieren Live Data (A3.1)
  const metricsPatterns = [
    "mrr", "cmrr", "arr", "revenue", "recognized", "reconoc", "billed", 
    "deferred", "diferid", "unbilled", "por factur", "facturación", 
    "facturacion", "ingresos", "ventas", "churn", "retention", 
    "retención", "retencion", "upsell", "downsell", "cross-sell",
    "new", "nuevo", "expansion", "expansión", "ndr", "quick ratio",

    "variación", "variacion", "cantidad", "cantidades", "quantity", "quantities"
  ];
  
  // Operaciones expandidas que requieren datos en vivo (A3.1)
  const operationPatterns = [
    "mostrar", "listar", "lista", "ver", "cuánto", "cuanto", "total",
    "suma", "ranking", "top", "mayor", "menor", "máximo", "maximo",
    "mínimo", "minimo", "promedio", "media", "por cliente", "por producto",

    "por industry", "por industria", "por país", "por pais", "por moneda",

  // Verificar patrones
  const hasTimePattern = timePatterns.some(pattern => {

  const hasMetrics = metricsPatterns.some(metric => q.includes(metric));
  const hasOperations = operationPatterns.some(op => q.includes(op));
  const hasOperational = operationalPatterns.some(op => q.includes(op));

  // Si la pregunta es conceptual (definición/reglas) y NO está pidiendo números/listados,
  // NO forzamos Live Data aunque mencione una métrica.
  const isConceptual = isConceptualQuery(question);
  if (isConceptual) {
    const hasNumericAsk =
      hasTimePattern ||
      hasOperations ||
      hasOperational ||
      /\b(cu[aá]nto|cu[aá]l|total|suma|monto|importe|valor|cuenta|n[uú]mero|serie|tendencia|hist[oó]rico)\b/i.test(q);

    if (!hasNumericAsk) {
      return false;
    }
  }
  
  // Regla de oro (A3.3): Si pide números/series/listados → Live Data
  if (hasMetrics) {
    return true; // Cualquier mención de métrica va a Live Data
  }
  
  if (hasTimePattern && hasOperations) {
    return true; // Tiempo + operaciones = Live Data
  }
  
  if (hasOperational) {
    return true; // Estados operativos = Live Data
  }
  
  // Patrones específicos de comparación/análisis
  if (q.includes("compar") || q.includes("vs") || q.includes("versus")) {
    return true;
  }
  
  return false;
}
