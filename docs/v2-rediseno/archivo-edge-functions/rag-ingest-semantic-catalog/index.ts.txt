import { serve } from "https://deno.land/std@0.190.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.49.5";

type IngestRequestBody = {
  limit?: number;
  entity?: string;
  source_type?: string;
  dry_run?: boolean;
  replace_existing?: boolean;
};

type SemanticCatalogRow = {
  id: string;
  holding_id: string;
  entity: string;
  column_name: string;
  business_name: string | null;
  description: string | null;
  unit: string | null;
  examples: unknown;
  created_at: string;
  updated_at: string;
};

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

function chunkArray<T>(arr: T[], size: number): T[][] {
  const out: T[][] = [];
  for (let i = 0; i < arr.length; i += size) {
    out.push(arr.slice(i, i + size));
  }
  return out;
}

function buildSemanticCatalogDoc(row: SemanticCatalogRow) {
  const title = `${row.entity}.${row.column_name}`;
  const examples = Array.isArray(row.examples) ? row.examples : [];

  const contentLines: string[] = [];
  contentLines.push("Catálogo semántico Sapira");
  contentLines.push(`Entidad: ${row.entity}`);
  contentLines.push(`Columna: ${row.column_name}`);
  if (row.business_name) contentLines.push(`Nombre de negocio: ${row.business_name}`);
  if (row.description) contentLines.push(`Descripción: ${row.description}`);
  if (row.unit) contentLines.push(`Unidad: ${row.unit}`);
  if (examples.length > 0) contentLines.push(`Ejemplos: ${JSON.stringify(examples)}`);

  const metadata = {
    title,
    source_type: "semantic_catalog",
    entity: row.entity,
    column_name: row.column_name,
    business_name: row.business_name,
    unit: row.unit,
  };

  return {
    source_type: "semantic_catalog",
    source_id: title,
    title,
    content: contentLines.join("\n"),
    metadata,
  };
}

async function createEmbeddings(apiKey: string, inputs: string[]): Promise<number[][]> {
  const res = await fetch("https://api.openai.com/v1/embeddings", {
    method: "POST",
    headers: {
      Authorization: `Bearer ${apiKey}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      model: "text-embedding-3-small",
      input: inputs,
    }),
  });

  if (!res.ok) {
    const text = await res.text().catch(() => "");
    throw new Error(`OpenAI embeddings error (${res.status}): ${text}`);
  }

  const json = await res.json();
  const data = json?.data;
  if (!Array.isArray(data) || data.length !== inputs.length) {
    throw new Error("OpenAI embeddings response shape mismatch");
  }

  const embeddings = data.map((d: any) => d?.embedding).filter(Boolean);
  if (embeddings.length !== inputs.length) {
    throw new Error("OpenAI embeddings response missing embedding");
  }

  return embeddings as number[][];
}

const handler = async (req: Request): Promise<Response> => {
  if (req.method === "OPTIONS") {
    return new Response(null, { headers: corsHeaders });
  }

  try {
    const supabaseUrl = Deno.env.get("SUPABASE_URL");
    const anonKey = Deno.env.get("SUPABASE_ANON_KEY");
    const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
    const openaiApiKey = Deno.env.get("OPENAI_API_KEY");

    if (!supabaseUrl || !anonKey || !serviceRoleKey) {
      return new Response(JSON.stringify({ error: "Server configuration error" }), {
        status: 500,
        headers: { "Content-Type": "application/json", ...corsHeaders },
      });
    }

    if (!openaiApiKey) {
      return new Response(JSON.stringify({ error: "Missing OPENAI_API_KEY" }), {
        status: 500,
        headers: { "Content-Type": "application/json", ...corsHeaders },
      });
    }

    const authHeader = req.headers.get("Authorization");
    if (!authHeader) {
      return new Response(JSON.stringify({ error: "Missing Authorization header" }), {
        status: 401,
        headers: { "Content-Type": "application/json", ...corsHeaders },
      });
    }

    const supabaseAuth = createClient(supabaseUrl, anonKey, {
      global: { headers: { Authorization: authHeader } },
    });

    const supabase = createClient(supabaseUrl, serviceRoleKey);

    const { data: authData, error: authError } = await supabaseAuth.auth.getUser();
    if (authError || !authData?.user) {
      return new Response(JSON.stringify({ error: "Invalid JWT" }), {
        status: 401,
        headers: { "Content-Type": "application/json", ...corsHeaders },
      });
    }

    const body = (await req.json().catch(() => ({}))) as IngestRequestBody;

    const limit = typeof body.limit === "number" ? body.limit : 500;
    const dryRun = body.dry_run === true;
    const replaceExisting = body.replace_existing !== false;
    const sourceType = typeof body.source_type === "string" && body.source_type.trim()
      ? body.source_type.trim()
      : "semantic_catalog";

    if (sourceType !== "semantic_catalog") {
      return new Response(JSON.stringify({ error: "Unsupported source_type" }), {
        status: 400,
        headers: { "Content-Type": "application/json", ...corsHeaders },
      });
    }

    const { data: holdingIdData, error: holdingErr } = await supabaseAuth.rpc(
      "get_current_user_holding_id",
    );
    const holdingId = holdingIdData as string | null;

    if (holdingErr || !holdingId) {
      return new Response(
        JSON.stringify({ error: "No holding_id for current user", details: holdingErr?.message }),
        {
          status: 403,
          headers: { "Content-Type": "application/json", ...corsHeaders },
        },
      );
    }

    let query = supabase
      .schema("analytics")
      .from("semantic_catalog")
      .select(
        "id, holding_id, entity, column_name, business_name, description, unit, examples, created_at, updated_at",
      )
      .eq("holding_id", holdingId)
      .order("updated_at", { ascending: false })
      .limit(limit);

    if (typeof body.entity === "string" && body.entity.trim()) {
      query = query.eq("entity", body.entity.trim());
    }

    const { data: rows, error: rowsError } = await query;
    if (rowsError) throw rowsError;

    const catalogRows = (rows || []) as SemanticCatalogRow[];
    const docs = catalogRows.map(buildSemanticCatalogDoc);

    const batches = chunkArray(docs, 50);

    let inserted = 0;
    let replaced = 0;

    for (const batch of batches) {
      const sourceIds = batch.map((d) => d.source_id);

      if (replaceExisting && !dryRun) {
        const { error: delErr } = await supabase
          .from("rag_documents")
          .delete()
          .eq("holding_id", holdingId)
          .eq("source_type", sourceType)
          .in("source_id", sourceIds);
        if (delErr) throw delErr;
        replaced += sourceIds.length;
      }

      const embeddings = dryRun
        ? batch.map(() => new Array(1536).fill(0))
        : await createEmbeddings(openaiApiKey, batch.map((d) => d.content));

      if (dryRun) {
        inserted += batch.length;
        continue;
      }

      const payload = batch.map((d, idx) => ({
        holding_id: holdingId,
        source_type: sourceType,
        source_id: d.source_id,
        content: d.content,
        metadata: d.metadata,
        embedding: embeddings[idx],
        updated_at: new Date().toISOString(),
      }));

      const { error: insErr } = await supabase.from("rag_documents").insert(payload);
      if (insErr) throw insErr;
      inserted += payload.length;
    }

    return new Response(
      JSON.stringify({
        success: true,
        holding_id: holdingId,
        source_type: sourceType,
        entity_filter: body.entity ?? null,
        dry_run: dryRun,
        replace_existing: replaceExisting,
        semantic_catalog_rows: catalogRows.length,
        rag_documents_inserted: inserted,
        rag_documents_deleted_estimate: replaceExisting ? replaced : 0,
      }),
      {
        status: 200,
        headers: { "Content-Type": "application/json", ...corsHeaders },
      },
    );
  } catch (error) {
    return new Response(JSON.stringify({ error: "Internal server error", details: error?.message ?? String(error) }), {
      status: 500,
      headers: { "Content-Type": "application/json", ...corsHeaders },
    });
  }
};

serve(handler);
