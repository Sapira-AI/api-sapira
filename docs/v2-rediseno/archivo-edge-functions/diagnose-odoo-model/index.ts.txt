// Cliente XML-RPC simplificado
class XmlRpcClient {
  private url: string;
  
  constructor(url: string) {
    this.url = url;
  }
  
  async methodCall(method: string, params: any[]): Promise<any> {
    console.log(`Enviando solicitud XML-RPC a ${this.url}:`);
    console.log(`Método: ${method}`);
    console.log(`Parámetros: ${JSON.stringify(params).substring(0, 200)}...`);
    
    const xml = this.createXmlRpcRequest(method, params);
    
    const response = await fetch(this.url, {
      method: 'POST',
      headers: {
        'Content-Type': 'text/xml',
      },
      body: xml,
    });
    
    if (!response.ok) {
      throw new Error(`XML-RPC request failed: ${response.status} ${response.statusText}`);
    }
    
    const responseText = await response.text();
    console.log(`Respuesta XML-RPC recibida (${responseText.length} bytes)`);
    
    return this.parseXmlRpcResponse(responseText);
  }
  
  private createXmlRpcRequest(method: string, params: any[]): string {
    let xml = '<?xml version="1.0"?>';
    xml += '<methodCall>';
    xml += `<methodName>${method}</methodName>`;
    xml += '<params>';
    
    for (const param of params) {
      xml += '<param>';
      xml += this.valueToXml(param);
      xml += '</param>';
    }
    
    xml += '</params>';
    xml += '</methodCall>';
    
    return xml;
  }
  
  private valueToXml(value: any): string {
    let xml = '<value>';
    
    if (value === null || value === undefined) {
      xml += '<nil/>';
    } else if (typeof value === 'string') {
      xml += `<string>${this.escapeXml(value)}</string>`;
    } else if (typeof value === 'number') {
      if (Number.isInteger(value)) {
        xml += `<int>${value}</int>`;
      } else {
        xml += `<double>${value}</double>`;
      }
    } else if (typeof value === 'boolean') {
      xml += `<boolean>${value ? '1' : '0'}</boolean>`;
    } else if (value instanceof Date) {
      xml += `<dateTime.iso8601>${value.toISOString()}</dateTime.iso8601>`;
    } else if (Array.isArray(value)) {
      xml += '<array><data>';
      for (const item of value) {
        xml += this.valueToXml(item);
      }
      xml += '</data>';
      xml += '</array>';
    } else if (typeof value === 'object') {
      xml += '<struct>';
      for (const [key, val] of Object.entries(value)) {
        xml += `<member><name>${this.escapeXml(key)}</name>`;
        xml += this.valueToXml(val);
        xml += '</member>';
      }
      xml += '</struct>';
    } else {
      // Fallback a string
      xml += `<string>${this.escapeXml(String(value))}</string>`;
    }
    
    xml += '</value>';
    return xml;
  }
  
  private escapeXml(str: string): string {
    return str
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;')
      .replace(/'/g, '&apos;');
  }
  
  private parseXmlRpcResponse(xml: string): any {
    console.log('Analizando respuesta XML-RPC con librería XML...');
    
    try {
      // Usar librería XML para parsear
      const parsed = parse(xml);
      console.log('✅ XML parseado exitosamente');
      
      // Navegar por la estructura XML-RPC estándar
      const methodResponse = parsed.methodResponse;
      if (!methodResponse) {
        throw new Error('No se encontró methodResponse');
      }
      
      // Verificar si hay fault
      if (methodResponse.fault) {
        const faultValue = methodResponse.fault.value;
        throw new Error(`XML-RPC Fault: ${JSON.stringify(faultValue)}`);
      }
      
      // Obtener el valor de la respuesta
      const params = methodResponse.params;
      if (!params || !params.param || !params.param.value) {
        throw new Error('Estructura de respuesta XML-RPC inválida');
      }
      
      const responseValue = params.param.value;
      
      // Convertir la estructura XML parseada a objetos JavaScript
      const result = this.convertXmlValueToJs(responseValue);
      
      console.log(`✅ Convertido a JavaScript: ${Array.isArray(result) ? result.length + ' elementos' : typeof result}`);
      return result;
      
    } catch (error) {
      console.error('❌ Error al parsear XML-RPC:', error.message);
      console.log('XML problemático (primeros 500 chars):', xml.substring(0, 500));
      return [];
    }
  }
  
  private convertXmlValueToJs(value: any): any {
    if (!value) return null;
    
    // Manejar diferentes tipos de valores XML-RPC
    if (value.int !== undefined) return parseInt(value.int, 10);
    if (value.i4 !== undefined) return parseInt(value.i4, 10);
    if (value.double !== undefined) return parseFloat(value.double);
    if (value.string !== undefined) return value.string;
    if (value.boolean !== undefined) return value.boolean === '1';
    
    // Manejar arrays
    if (value.array && value.array.data && value.array.data.value) {
      const arrayValues = Array.isArray(value.array.data.value) 
        ? value.array.data.value 
        : [value.array.data.value];
      
      return arrayValues.map(v => this.convertXmlValueToJs(v));
    }
    
    // Manejar structs (objetos)
    if (value.struct && value.struct.member) {
      const obj = {};
      const members = Array.isArray(value.struct.member) 
        ? value.struct.member 
        : [value.struct.member];
      
      for (const member of members) {
        if (member.name && member.value !== undefined) {
          obj[member.name] = this.convertXmlValueToJs(member.value);
        }
      }
      return obj;
    }
    
    // Valor directo
    return value;
  }
}

            console.log('Enviando solicitud read con parámetros:', {
              db: connection.database_name,
              uid,
              model,
              method: 'read',
              record_id
            });
            
            const readResponse = await objectClient.methodCall('execute_kw', [
              connection.database_name,
              uid,
              connection.api_key,
              model,
              'read',
              [[record_id]]  // Array con el ID específico
              // Sin especificar fields para obtener todos los campos
            ]);
            
            console.log('Respuesta read recibida, tipo:', typeof readResponse);
            console.log('Es array?', Array.isArray(readResponse));
            
            if (Array.isArray(readResponse) && readResponse.length > 0) {
              // Tenemos el registro específico
              const record = readResponse[0];
              console.log(`Registro ${record_id} obtenido exitosamente`);
              console.log('Campos disponibles:', Object.keys(record).length);
              
              // Mostrar información detallada del registro
              console.log('Información del registro:');
              console.log('- ID:', record.id);
              if (record.name) console.log('- Nombre:', record.name);
              if (record.display_name) console.log('- Display Name:', record.display_name);
              
              // Para facturas, mostrar campos específicos
              if (model === 'account.move') {
                console.log('Campos específicos de factura:');
                console.log('- partner_id:', record.partner_id);
                console.log('- amount_total:', record.amount_total);
                console.log('- state:', record.state);
                console.log('- invoice_date:', record.invoice_date);
              }
              
              // Para partners, mostrar campos específicos
              if (model === 'res.partner') {
                console.log('Campos específicos de partner:');
                console.log('- name:', record.name);
                console.log('- email:', record.email);
                console.log('- is_company:', record.is_company);
                console.log('- customer_rank:', record.customer_rank);
                console.log('- supplier_rank:', record.supplier_rank);
              }
              
              result = {
                success: true,
                record: record,
                record_id: record_id,
                model: model,
                fields_count: Object.keys(record).length,
                message: `Registro ${record_id} obtenido exitosamente del modelo ${model}`
              };
            } else if (Array.isArray(readResponse) && readResponse.length === 0) {
              result = {
                error: `No se encontró el registro con ID ${record_id} en el modelo ${model}`,
                record_id: record_id,
                model: model,
                recommendation: 'Verifica que el ID existe y que tienes permisos para acceder a él'
              };
            } else if (typeof readResponse === 'string') {
              console.log('Respuesta como string:', readResponse);
              result = {
                error: 'La API de Odoo devolvió un mensaje de texto en lugar del registro',
                message: readResponse,
                record_id: record_id,
                recommendation: 'Verifica que el usuario tenga permisos suficientes para acceder a este registro'
              };
            } else {
              console.log('Respuesta inesperada:', readResponse);
              result = {
                error: 'Respuesta inesperada de la API de Odoo',
                response: readResponse,
                record_id: record_id,
                recommendation: 'Verifica la conexión y los permisos'
              };
            }
          } catch (recordError) {
            console.error(`Error al obtener registro ${record_id}:`, recordError);
            result = {
              error: `Error al obtener registro ${record_id}: ${recordError.message}`,
              record_id: record_id,
              model: model,
              stack: recordError.stack
            };
          }
        }
      } else if (action === 'count_by_type') {
        console.log(`Contando registros por tipo en el modelo ${model}...`);
        try {
          console.log('Enviando solicitud read_group con parámetros:', {
            db: connection.database_name,
            uid,
            model,
            method: 'read_group'
          });
          
          const countResponse = await objectClient.methodCall('execute_kw', [
            connection.database_name,
            uid,
            connection.api_key,
            model,
            'read_group',
            [[]],
            {'fields': ['move_type'], 'groupby': ['move_type']}
          ]);
          
          console.log('Respuesta read_group recibida, tipo:', typeof countResponse);
          console.log('Es array?', Array.isArray(countResponse));
          
          if (Array.isArray(countResponse)) {
            console.log(`Se encontraron ${countResponse.length} tipos de registros`);
            console.log('Tipos encontrados:', countResponse);
          } else if (typeof countResponse === 'string') {
            console.log('Respuesta como string:', countResponse);
            result = {
              error: 'Respuesta inesperada de Odoo',
              raw_response: countResponse
            };
            return;
          }
          
          result = countResponse;          
        } catch (countError) {
          console.error('Error al contar registros por tipo:', countError);
          result = {
            error: `Error al contar registros por tipo: ${countError.message}`,
            stack: countError.stack
          };
        }
      }
      
      // Devolver respuesta exitosa
      console.log('Diagnóstico completado');
      return new Response(
        JSON.stringify({ 
          success: true, 
          model,
          action,
          result
        }),
        { headers: responseHeaders }
      );
    } catch (xmlrpcError) {
      console.error('Error general en XML-RPC:', xmlrpcError);
      
      // Manejo especial para list_models: si falla por permisos, usar lista predefinida
      if (action === 'list_models' && xmlrpcError.message && xmlrpcError.message.includes('ir.model')) {
        console.log('🔄 Error de permisos en list_models, usando lista predefinida...');
        
        const fallbackModels = [
          { 
            model: 'account.move', 
            name: 'Asiento contable/Factura',
            info: 'Facturas de cliente y proveedor, notas de crédito',
            description: 'Modelo principal para facturas y asientos contables'
          },
          { 
            model: 'account.move.line', 
            name: 'Línea de asiento contable',
            info: 'Líneas individuales de facturas',
            description: 'Cada línea de una factura o asiento contable'
          },
          { 
            model: 'res.partner', 
            name: 'Contacto/Cliente/Proveedor',
            info: 'Clientes, proveedores y contactos',
            description: 'Información de clientes, proveedores y contactos'
          },
          { 
            model: 'product.product', 
            name: 'Producto (Variante)',
            info: 'Productos específicos con variantes',
            description: 'Productos individuales con sus variantes'
          },
          { 
            model: 'product.template', 
            name: 'Plantilla de producto',
            info: 'Plantillas base de productos',
            description: 'Plantillas base para crear variantes de productos'
          },
          { 
            model: 'account.account', 
            name: 'Cuenta contable',
            info: 'Plan de cuentas contables',
            description: 'Cuentas del plan contable'
          },
          { 
            model: 'account.journal', 
            name: 'Diario contable',
            info: 'Diarios de ventas, compras, etc.',
            description: 'Diarios para organizar asientos contables'
          },
          { 
            model: 'account.payment', 
            name: 'Pago',
            info: 'Pagos de clientes y a proveedores',
            description: 'Registros de pagos y cobros'
          }
        ];
        
        console.log(`✅ Devolviendo lista predefinida: ${fallbackModels.length} modelos`);
        
        return new Response(
          JSON.stringify({ 
            success: true, 
            model,
            action,
            result: fallbackModels,
            fallback: true,
            message: 'Lista predefinida de modelos (permisos insuficientes para acceder a ir.model)'
          }),
          { headers: responseHeaders }
        );
      }
      
      return new Response(
        JSON.stringify({ 
          success: false, 
          message: `Error en comunicación XML-RPC con Odoo: ${xmlrpcError.message}`,
          error_details: String(xmlrpcError)
        }),
        { headers: responseHeaders, status: 500 }
      );
    }
  } catch (error) {
    // Manejar errores generales
    console.error('Error general en diagnose-odoo-model:', error);
    
    return new Response(
      JSON.stringify({ 
        success: false, 
        message: `Error al procesar la solicitud: ${error.message}`,
        stack: error.stack
      }),
      { headers: responseHeaders, status: 500 }
    );
  }
});
